cmake_minimum_required(VERSION 3.8)
project(pb_omni_pid_pursuit_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_util REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/omni_pid_pursuit_controller.cpp src/pid.cpp)
include_directories(include)
ament_target_dependencies(${PROJECT_NAME} nav2_core nav2_util nav2_costmap_2d nav2_msgs rclcpp geometry_msgs nav_msgs pluginlib tf2 tf2_geometry_msgs visualization_msgs)
pluginlib_export_plugin_description_file(nav2_core pb_omni_pid_pursuit_controller.xml)
install(TARGETS ${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_libraries(${PROJECT_NAME})
ament_export_include_directories(include)
ament_package()
